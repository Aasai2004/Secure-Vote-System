  apiKey: "YOUR_KEY",
  authDomain: "YOUR_DOMAIN",
  projectId: "YOUR_ID",
};

const app = initializeApp(firebaseConfig);
const db = getFirestore(app);

export { db };
🔥 NOW CONNECT YOUR PAGES
🧑‍💼 ADMIN PAGE (Store Data)
import { db } from "./firebase-config.js";
import { collection, addDoc } from "https://www.gstatic.com/firebasejs/10.7.1/firebase-firestore.js";

async function addVoter() {
  const aadhar = document.getElementById("aadhar").value;

  await addDoc(collection(db, "Voters"), {
    aadhar: aadhar
  });

  alert("Voter Added!");
}
👤 LOGIN PAGE (Check Aadhar)
import { db } from "./firebase-config.js";
import { collection, getDocs } from "https://www.gstatic.com/firebasejs/10.7.1/firebase-firestore.js";

async function checkAadhar() {
  const input = document.getElementById("aadhar").value;

  const querySnapshot = await getDocs(collection(db, "Voters"));

  let found = false;

  querySnapshot.forEach((doc) => {
    if (doc.data().aadhar === input) {
      found = true;
    }
  });

  if (found) {
    window.location.href = "verify.html";
  } else {
    alert("Aadhar not registered");
  }
}
🗳️ VOTING PAGE (Store Vote)
import { db } from "./firebase-config.js";

📊 RESULT PAGE (Show Votes)
import { db } from "./firebase-config.js";
import { collection, getDocs } from "https://www.gstatic.com/firebasejs/10.7.1/firebase-firestore.js";

async function loadResults() {
  const snapshot = await getDocs(collection(db, "Votes"));

  let countA = 0;

  snapshot.forEach((doc) => {
    if (doc.data().candidate === "A") {
      countA++;
    }
  });

  document.getElementById("result").innerText = countA;
}
